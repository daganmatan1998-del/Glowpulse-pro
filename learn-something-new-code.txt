  set: new Set(),
  load(){
    try{
      const raw = localStorage.getItem(KEY);
      if (raw) this.set = new Set(JSON.parse(raw) || []);
    }catch(e){ /* a corrupt or blocked store must never stop the site loading */ }
    // Drop entries for topics that no longer exist, so the count cannot exceed
    // the library after topics are edited.
    const titles = new Set(TOPICS.map(t => t.t));
    for (const t of [...this.set]) if (!titles.has(t)) this.set.delete(t);
  },
  save(){ try{ localStorage.setItem(KEY, JSON.stringify([...this.set])); }catch(e){} },
  has(t){ return this.set.has(t.t); },
  toggle(t){ this.set.has(t.t) ? this.set.delete(t.t) : this.set.add(t.t); this.save(); paintProgress(); },
  count(){ return this.set.size; }
};

/* ==========================================================================
   THE ROULETTE

   Pressing Start spins a reel of topics past the centre and lands on one.

   The winner is chosen up front and the reel is built so that winner sits at
   a known row; the animation then eases from 0 to that exact distance. Doing
   it the other way round — spinning freely and deciding where to stop — means
   either snapping at the end or landing between two rows, and both are
   immediately visible.

   Tune the feel here:
      SPIN_ROWS   how many topics fly past (the sense of searching hundreds)
      SPIN_MS     total duration, fast start to full stop
      EASE        the deceleration curve; higher power = harder braking
   ========================================================================== */
const SPIN_ROWS = 78;
const SPIN_MS   = 4600;
const EASE      = t => 1 - Math.pow(1 - t, 5.2);   // very fast → … → stop

let spinning = false;

function spin(onDone){
  if (spinning) return;
  spinning = true;

  const wrap = $('stagewrap');
  // A zero height — a hidden tab, a font still loading — would collapse every
  // row and the reel would appear not to move at all.
  const rowH = Math.max(160, wrap.clientHeight);

  // The winner. Chosen from the whole library, never biased toward the rows
  // shown first, and it becomes the current topic so the arrows carry on from
  // wherever the reel stopped.
  const winner = Math.floor(Math.random() * TOPICS.length);
  state.i = state.order.indexOf(winner);

  // Build the reel: SPIN_ROWS random topics, with the winner placed last.
  const rows = [];
  for (let i = 0; i < SPIN_ROWS; i++) rows.push(Math.floor(Math.random() * TOPICS.length));
  rows.push(winner);

  const reel = document.createElement('div');
  reel.className = 'reel';
  reel.innerHTML = rows.map(i =>
    `<div class="row" style="height:${rowH}px">${esc(TOPICS[i].t)}</div>`).join('');

  const mask = document.createElement('div');
  mask.className = 'reelmask';
  mask.appendChild(reel);

  // Take every topic out of the stage rather than setting opacity to 0.
  // Hiding relies on stacking, masks and opacity all behaving; removing does
  // not, so there is no browser on which the old topic can survive the spin.
  wrap.querySelectorAll('.topic').forEach(el => el.remove());
  wrap.appendChild(mask);

  const distance = SPIN_ROWS * rowH;          // stops with the winner centred
  const t0 = performance.now();
  let lastY = 0;

  function frame(now){
    const t = Math.min(1, (now - t0) / SPIN_MS);
    const y = EASE(t) * distance;

    // Motion blur from the actual frame-to-frame speed, which is what turns
    // the fast section into a smear and the last rows into readable text.
    const speed = Math.abs(y - lastY);
    lastY = y;
    const blur = Math.min(14, speed * 0.075);

    reel.style.transform = `translate3d(0,${-y}px,0)`;
    reel.style.filter = blur > 0.4 ? `blur(${blur.toFixed(2)}px)` : '';

    if (t < 1) return requestAnimationFrame(frame);

    mask.remove();
    spinning = false;
    onDone(TOPICS[winner]);
  }
  requestAnimationFrame(frame);
}

}

function runSearch(term){
  const q = term.trim().toLowerCase();
  // Rank title matches above category and hook matches, so typing a subject
  // still finds the topic named after it first.
  hits = (q ? TOPICS.map(t => {
      const title = t.t.toLowerCase(), cat = t.c.toLowerCase(), hook = t.hook.toLowerCase();
      let score = 0;
      if (title.includes(q)) score = title.startsWith(q) ? 100 : 60;
      else if (cat.includes(q)) score = 30;
      else if (hook.includes(q)) score = 15;
      return { t, score };
    }).filter(r => r.score > 0).sort((a, b) => b.score - a.score).map(r => r.t)
    : TOPICS.slice(0, 8));

  selIndex = 0;
  $('results').innerHTML = hits.length
    ? hits.slice(0, 40).map((t, i) =>
        `<button class="result ${i === 0 ? 'sel' : ''}" data-i="${i}">
           <b>${esc(t.t)}</b><small>${esc(t.c)} · ${t.m} min</small></button>`).join('')
    : '<div class="empty" style="padding:34px">No topic matches that.</div>';

  $('results').querySelectorAll('.result').forEach(r => {
    r.onclick = () => { closeSearch(); openTopic(hits[+r.dataset.i]); };
  });
}
function moveSel(d){
  const rows = $('results').querySelectorAll('.result');
  if (!rows.length) return;
  rows[selIndex]?.classList.remove('sel');
  selIndex = (selIndex + d + rows.length) % rows.length;
  rows[selIndex].classList.add('sel');
  rows[selIndex].scrollIntoView({ block:'nearest' });
}

addEventListener('keydown', e => {
  if ($('studio').classList.contains('on')){
    if (e.key === 'Escape') Studio.close();
    return;                       // the studio owns the keyboard while it is open
  }
  const searching = $('veil').classList.contains('on');

  if (searching){
    if (e.key === 'Escape') return closeSearch();
    if (e.key === 'ArrowDown'){ e.preventDefault(); return moveSel(1); }
    if (e.key === 'ArrowUp'){ e.preventDefault(); return moveSel(-1); }
    if (e.key === 'Enter' && hits.length){ closeSearch(); openTopic(hits[selIndex]); }
    return;
  }

  // Never hijack typing.
  if (e.target.tagName === 'INPUT') return;

  if (e.key === '/'){ e.preventDefault(); return openSearch(); }
  if (e.key === 'Escape' && state.view === 'topic') return go(state.started ? 'learn' : 'explore');
  if (e.key.toLowerCase() === 'r'){ e.preventDefault(); return surprise(); }

  if (state.view !== 'learn') return;
  if (e.key === 'ArrowDown'){ e.preventDefault(); state.started ? move(1) : start(); }
  if (e.key === 'ArrowUp'){ e.preventDefault(); if (state.started) move(-1); }
  if (e.key === 'Enter'){ e.preventDefault(); state.started ? openTopic(current()) : start(); }
});


/* =========================================================================
   VIDEO VAULT
   Clips are video blobs, which are far too big for localStorage, so they live
   in IndexedDB. Every call resolves rather than throws: a browser in private
   mode with storage blocked must degrade to "you cannot save clips" and leave
   the rest of the site working.
   ========================================================================= */
const Vault = {
  db:null, broken:false,
  open(){
    if (this.db) return Promise.resolve(this.db);
    if (this.broken || !window.indexedDB) return Promise.resolve(null);
    return new Promise(resolve => {
      let req;
      try{ req = indexedDB.open('lsn.videos', 1); }
      catch(e){ this.broken = true; return resolve(null); }

      req.onupgradeneeded = () => {
        const db = req.result;
        if (!db.objectStoreNames.contains('clips'))
          db.createObjectStore('clips', { keyPath:'id', autoIncrement:true });
      };
      req.onsuccess = () => { this.db = req.result; resolve(this.db); };
      req.onerror   = () => { this.broken = true; resolve(null); };
      req.onblocked = () => resolve(null);
    });
  },
  async tx(mode){
    const db = await this.open();
    return db ? db.transaction('clips', mode).objectStore('clips') : null;
  },
  async add(rec){
    const store = await this.tx('readwrite');
    if (!store) return null;
    return new Promise(resolve => {
      const r = store.add(rec);
      r.onsuccess = () => resolve(r.result);
      r.onerror   = () => resolve(null);
    });
  },
  async all(){
    const store = await this.tx('readonly');
    if (!store) return [];
    return new Promise(resolve => {
      const r = store.getAll();
      // Newest first: the clip you just made should be the one you see.
      r.onsuccess = () => resolve((r.result || []).sort((a, b) => b.at - a.at));
      r.onerror   = () => resolve([]);
    });
  },
  async del(id){
    const store = await this.tx('readwrite');
    if (!store) return;
    return new Promise(resolve => { const r = store.delete(id); r.onsuccess = r.onerror = () => resolve(); });
  }
};

/* ---- small helpers ------------------------------------------------------ */
function clock(sec){
  sec = Math.max(0, Math.round(sec));
  return Math.floor(sec / 60) + ':' + String(sec % 60).padStart(2, '0');
}
function weight(bytes){
  if (bytes < 1024 * 1024) return Math.max(1, Math.round(bytes / 1024)) + ' KB';
  return (bytes / 1048576).toFixed(bytes > 10485760 ? 0 : 1) + ' MB';
}
function when(ms){
  const d = new Date(ms), now = Date.now();
  if (now - ms < 86400000 && d.getDate() === new Date().getDate())
    return 'Today, ' + d.toLocaleTimeString([], { hour:'numeric', minute:'2-digit' });
  return d.toLocaleDateString([], { day:'numeric', month:'short', year:'numeric' });
}
let toastTimer = 0;
function toast(msg){
  let el = $('toast');
  if (!el){
    el = document.createElement('div');
    el.id = 'toast'; el.className = 'toast';
    document.body.appendChild(el);
  }
  el.textContent = msg;
  el.classList.add('on');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => el.classList.remove('on'), 2600);
}

/* =========================================================================
   THE STUDIO — camera, recorder, and the save panel.
   ========================================================================= */
const Studio = {
  stream:null, rec:null, chunks:[], blob:null, url:null,
  mode:'off',            // off | live | rec | review
  t0:0, tick:0, topic:null, counting:false,

  MAX_MS: 10 * 60 * 1000,   // a hard ceiling, or one forgotten tab fills the disk

  /* The first mime type the browser will actually record in. Chrome and
     Firefox want webm, Safari only ever offers mp4, so asking is the only
     portable way to find out. */
  mime(){
    const wanted = ['video/webm;codecs=vp9,opus','video/webm;codecs=vp8,opus',
                    'video/webm','video/mp4;codecs=h264,aac','video/mp4'];
    for (const m of wanted)
      if (window.MediaRecorder && MediaRecorder.isTypeSupported(m)) return m;
    return '';
  },

  open(topic){
    this.topic = topic || null;
    $('studio').classList.add('on');
    $('stTitle').textContent = topic ? 'Explain it in your own words' : 'Record an explanation';
    $('stSub').textContent = topic
      ? '“' + topic.t + '” — say it as if to someone who has never heard of it. Where you stumble is the part you have not learned yet.'
      : 'Pick the topic below when you are done. Say it as if to someone who has never heard of it.';
    this.reset();
    this.fillTopics();
    document.body.style.overflow = 'hidden';
  },

  close(){
    this.stopStream();
    $('studio').classList.remove('on');
    document.body.style.overflow = '';
    this.reset();
  },

  reset(){
    this.chunks = []; this.blob = null;
    if (this.url){ URL.revokeObjectURL(this.url); this.url = null; }
    clearInterval(this.tick);

    const v = $('stVideo');
    v.srcObject = null; v.removeAttribute('src'); v.controls = false; v.muted = true;
    v.classList.add('mirror');

    $('stMsg').style.display = '';
    $('stRec').classList.remove('on');
    $('stDot').classList.remove('live');
    $('stDot').disabled = false;
    $('stSave').style.display = 'none';
    $('stTime').textContent = '0:00';
    $('stHint').textContent = 'Press the red button to turn on your camera.';
    this.mode = 'off';
    this.counting = false;
  },

  stopStream(){
    if (this.rec && this.rec.state !== 'inactive'){ try{ this.rec.stop(); }catch(e){} }
    this.rec = null;
    if (this.stream){ this.stream.getTracks().forEach(t => t.stop()); this.stream = null; }
    clearInterval(this.tick);
  },

  message(title, body){
    $('stMsg').style.display = '';
    $('stMsg').innerHTML = '<b>' + esc(title) + '</b>' + (body ? '<span>' + esc(body) + '</span>' : '');
  },

  async camera(){
    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia){
      this.message('No camera available here',
        'This browser will not hand over a camera on a local file. You can still upload a video you recorded elsewhere.');
      $('stHint').textContent = 'Use “Upload a video instead”.';
      return false;
    }
    this.message('Asking for your camera…', 'Allow access in the prompt at the top of the window.');
    try{
      this.stream = await navigator.mediaDevices.getUserMedia({
        video:{ width:{ ideal:1280 }, height:{ ideal:720 }, facingMode:'user' },
        audio:{ echoCancellation:true, noiseSuppression:true }
      });
    }catch(e){
      const denied = e && (e.name === 'NotAllowedError' || e.name === 'SecurityError');
      this.message(denied ? 'Camera permission was refused' : 'No camera could be opened',
        denied ? 'Allow it in your browser’s site settings, or upload a video instead.'
               : 'Nothing is plugged in, or another app is already using it. You can upload a video instead.');
      $('stHint').textContent = 'Use “Upload a video instead”.';
      return false;
    }

    const v = $('stVideo');
    v.srcObject = this.stream; v.muted = true; v.classList.add('mirror');
    try{ await v.play(); }catch(e){}
    $('stMsg').style.display = 'none';
    $('stHint').textContent = 'Looking good. Press the button again to start recording.';
    this.mode = 'live';
    return true;
  },

  /* A three-second run-in, because starting to talk the instant a camera
     turns on is how you get twenty clips that begin with "um". */
  countdown(){
    if (this.counting) return;
    this.counting = true;
    let n = 3;
    this.message(String(n), 'Get ready');
    const step = setInterval(() => {
      n--;
      if (n > 0){ this.message(String(n), 'Get ready'); return; }
      clearInterval(step);
      this.counting = false;
      $('stMsg').style.display = 'none';
      this.record();
    }, 800);
  },

  record(){
    const type = this.mime();
    if (!window.MediaRecorder || !type){
      this.message('This browser cannot record video',
        'MediaRecorder is not available. You can still upload a video you made elsewhere.');
      return;
    }

    this.chunks = [];
    try{ this.rec = new MediaRecorder(this.stream, { mimeType:type, videoBitsPerSecond:2_500_000 }); }
    catch(e){ this.rec = new MediaRecorder(this.stream); }

    this.rec.ondataavailable = e => { if (e.data && e.data.size) this.chunks.push(e.data); };
    this.rec.onstop = () => this.review(new Blob(this.chunks, { type:this.rec.mimeType || type }),
                                        (Date.now() - this.t0) / 1000);

    this.rec.start(250);
    this.t0 = Date.now();
    this.mode = 'rec';

    $('stDot').classList.add('live');
    $('stDot').setAttribute('aria-label', 'Stop recording');
    $('stRec').classList.add('on');
    $('stHint').textContent = 'Recording. Press the square to stop.';

    clearInterval(this.tick);
    this.tick = setInterval(() => {
      const ms = Date.now() - this.t0;
      $('stTime').textContent = clock(ms / 1000);
      if (ms >= this.MAX_MS){ toast('Ten minutes is the limit — saved what you had.'); this.stop(); }
    }, 250);
  },

  stop(){
    clearInterval(this.tick);
    if (this.rec && this.rec.state !== 'inactive') this.rec.stop();
  },

  /* Shows the clip back, whether it came from the camera or from a file. */
  review(blob, seconds){
    this.stopStream();
    this.blob = blob;
    this.dur = seconds;

    if (this.url) URL.revokeObjectURL(this.url);
    this.url = URL.createObjectURL(blob);

    const v = $('stVideo');
    v.srcObject = null;
    v.src = this.url;
    v.muted = false; v.controls = true; v.classList.remove('mirror');

    $('stMsg').style.display = 'none';
    $('stRec').classList.remove('on');
    $('stDot').classList.remove('live');
    $('stDot').disabled = true;
    $('stSave').style.display = '';
    $('stHint').textContent = 'Watch it back. Keep it, or go again.';
    this.mode = 'review';

    // A recorded webm often has no duration in its header, so the timer we
    // kept while recording is the honest number. An uploaded file does know.
    if (!seconds){
      v.onloadedmetadata = () => {
        if (isFinite(v.duration) && v.duration > 0) this.dur = v.duration;
      };
    }
  },

  fillTopics(){
    const sel = $('stTopic');
    const groups = {};
    for (const t of TOPICS) (groups[t.c] = groups[t.c] || []).push(t);

    sel.innerHTML = Object.keys(groups).sort().map(c =>
      `<optgroup label="${esc(c)}">${groups[c].map(t =>
        `<option value="${esc(t.t)}">${esc(t.t)}</option>`).join('')}</optgroup>`).join('');

    if (this.topic) sel.value = this.topic.t;
  },

  async save(){
    if (!this.blob) return;
    const title = $('stTopic').value;
    const t = TOPICS.find(x => x.t === title);

    const id = await Vault.add({
      topic: title,
      cat:   t ? t.c : '',
      at:    Date.now(),
      dur:   this.dur || 0,
      size:  this.blob.size,
      type:  this.blob.type || 'video/webm',
      blob:  this.blob
    });

    if (id === null){
      toast('This browser will not let the site store video.');
      return;
    }

    // Explaining it is a stronger signal of having learned it than a checkbox,
    // so saving a clip ticks the topic off if it was not already.
    if (t && !Progress.has(t)) Progress.toggle(t);

    this.close();
    toast('Saved to your gallery.');
    if (state.view === 'gallery') renderGallery();
    if (state.view === 'topic' && state.open) fillMine(state.open);
  }
};

/* =========================================================================
   THE GALLERY
   ========================================================================= */
let clipUrls = [];
function dropUrls(){ clipUrls.forEach(URL.revokeObjectURL); clipUrls = []; }

async function renderGallery(){
  dropUrls();
  const list = await Vault.all();

  $('galEmpty').style.display = list.length ? 'none' : '';
  $('clips').style.display    = list.length ? '' : 'none';

  const bytes = list.reduce((a, c) => a + (c.size || 0), 0);
  $('galInfo').textContent = list.length
    ? `${list.length} recording${list.length === 1 ? '' : 's'} · ${weight(bytes)} · stored in this browser`
    : '';

  $('clips').innerHTML = list.map(c => {
    const url = URL.createObjectURL(c.blob);
    clipUrls.push(url);
    const known = TOPICS.some(t => t.t === c.topic);
    return `<div class="clip">
      <video src="${url}" controls preload="metadata" playsinline></video>
      <div class="body">
        <h3>${esc(c.topic || 'Untitled')}</h3>
        <div class="sub">
          ${c.cat ? `<span class="chip cat">${esc(c.cat)}</span>` : ''}
          <span>${when(c.at)}</span>${c.dur ? `<span>· ${clock(c.dur)}</span>` : ''}
          <span>· ${weight(c.size || 0)}</span>
        </div>
        <div class="acts">
          ${known ? `<button data-open="${esc(c.topic)}">Re-read it</button>` : ''}
          <a href="${url}" download="${esc((c.topic || 'explanation').replace(/[^\w\- ]+/g, '')).slice(0, 60)}.${(c.type || '').includes('mp4') ? 'mp4' : 'webm'}">Download</a>
          <button class="danger" data-del="${c.id}">Delete</button>
        </div>
      </div>
    </div>`;
  }).join('');

  $('clips').querySelectorAll('[data-open]').forEach(b => {
    b.onclick = () => { const t = TOPICS.find(x => x.t === b.dataset.open); if (t) openTopic(t); };
  });

  // Two-step delete rather than a confirm box: the first press arms it, the
  // second does it, and clicking anywhere else disarms.
  $('clips').querySelectorAll('[data-del]').forEach(b => {
    b.onclick = async () => {
      if (!b.classList.contains('arm')){
        $('clips').querySelectorAll('.arm').forEach(o => { o.classList.remove('arm'); o.textContent = 'Delete'; });
        b.classList.add('arm'); b.textContent = 'Delete for good?';
        setTimeout(() => { b.classList.remove('arm'); b.textContent = 'Delete'; }, 4000);
        return;
      }
      await Vault.del(+b.dataset.del);
      renderGallery();
      toast('Deleted.');
    };
  });
}

/* Recordings you already made for the topic you are reading. Filled in after
   the article renders, because reading them is a trip to IndexedDB. */
async function fillMine(t){
  const box = $('mine');
  if (!box) return;
  const list = (await Vault.all()).filter(c => c.topic === t.t);
  if (!list.length){ box.closest('.sec').style.display = 'none'; return; }

  box.closest('.sec').style.display = '';
  box.innerHTML = list.map(c => {
    const url = URL.createObjectURL(c.blob);
    clipUrls.push(url);
    return `<div><video src="${url}" controls preload="metadata" playsinline></video>
      <div class="sub" style="color:var(--ink-3);font-size:12.5px;margin-top:6px">
        ${when(c.at)}${c.dur ? ' · ' + clock(c.dur) : ''}</div></div>`;
  }).join('');
}
